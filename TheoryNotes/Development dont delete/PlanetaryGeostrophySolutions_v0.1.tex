\citet{peterson&callies26} (see also \citealt{peterson&callies23}) consider how bottom-intensified vertical mixing induces a basin-scale circulation in various simple non-flat-bottomed basins.
They write equations for a bottom Ekman layer and couple them to an equation for the geostrophic contour vorticity balance, in the parlance of \citet{waldman&giordani23}, which is a type of barotropic vorticity equation.

\section{Governing Equations}
The flow satisfies the (non-dimensional) planetary-geostrophic and hydrostatic equations:
\begin{align}
- f v & = - \frac{\partial p}{\partial x} + \epsilon^2 \frac{\partial}{\partial z} \left( \nu \frac{\partial u}{\partial z} \right), \label{eqn:u_mom} \\
f  u & = - \frac{\partial p}{\partial y} + \epsilon^2 \frac{\partial}{\partial z} \left( \nu \frac{\partial v}{\partial z} \right),  \label{eqn:v_mom} \\
\frac{\partial p}{\partial z} & = b
\end{align}
(see Eqs.~(9)--(11) of \citealt{peterson&callies26} and Eq.~(3.26) of \citealt{klinger&haine19}).
The flow is non-divergent:
\begin{align}
\nabla \cdot {\mathbf u} & = 0 ,
\end{align}
and the buoyancy field satisfies the advection-diffusion equation
\begin{align}
\mu \rho \left( \frac{\partial b}{\partial t} + {\mathbf u} \cdot \nabla b \right) = \epsilon^2 \frac{\partial }{\partial z} \left( \kappa \frac{\partial b}{\partial z} \right) .
\label{eqn:b}
\end{align}
(see Eqs.~(11)--(13) of \citealt{peterson&callies26}).
The boundary conditions on buoyancy are:
\begin{align}
\text{Surface~}z = 0:
b &= 0
\label{eqn:sfc_b_bc}
 \\
\text{Bottom~}z = -H:
\frac{\partial b}{\partial z} &= 0 ,
\label{eqn:bot_b_bc}
\end{align}
and the initial condition is $b = z$.

The diffusivity and viscosity are given by
\begin{align}
\kappa(z) & = \kappa_0 + \exp \left( - \alpha (z + H) \right) , \label{eqn:kappa_def} \\
\nu & = \text{constant}
\end{align}
\citet{peterson&callies26} Eq.~(17).

\section{Vertically-Integrated Flow}
The vertically-integrated flow $(U, V)$ is given by:
\begin{align}
U & \equiv \int_{-H}^{0} u \; dz = - \int_{-H}^0 z \partial_z u \; dz , \nonumber \\
V & \equiv \int_{-H}^{0} v \; dz = - \int_{-H}^0 z \partial_z v \; dz ,
\label{eqn:UV_defs}
\end{align}
exploiting the bottom boundary condition on $(u, v)$.
The equations for $(U, V)$ come from vertically integrating the momentum equations (\ref{eqn:u_mom}), (\ref{eqn:v_mom}):
\begin{align}
- f V & = -  \int_{-H}^{0} \frac{\partial p}{\partial x} \; dz + \epsilon^2   \int_{-H}^{0} \frac{\partial}{\partial z} \left( \nu \frac{\partial u}{\partial z} \right) \; dz \label{eqn:U_mom0} \\
f  U & = -  \int_{-H}^{0} \frac{\partial p}{\partial y} \; dz + \epsilon^2  \int_{-H}^{0} \frac{\partial}{\partial z} \left( \nu \frac{\partial v}{\partial z} \right) \; dz . \label{eqn:V_mom0}
\end{align}
The depth-integrated pressure gradient in the $x$ direction is:
\begin{align}
 \int_{-H}^{0} \frac{\partial p}{\partial x} \; dz & = - \frac{\partial H}{\partial x} p(x,z = -H) +  \frac{\partial }{\partial x} \int_{-H}^0 p \; dz , \\
 & = - p_b \frac{\partial H}{\partial x} + \frac{\partial }{\partial x} \left[ p_b H - \int_{-H}^0 z \frac{\partial p}{\partial z}  \right], \\
  & = H \frac{\partial p_b}{\partial x} - \frac{\partial }{\partial x} \int_{-H}^0 z b \; dz ,
 \end{align}
 using the Liebniz integration formula\footnote{The Liebniz integration formula reads:
 \begin{align}
 \partial_x \left( \int_{a(x)}^{b(x)} f(x,y) \; dy \right) = \partial_x b f(x,b(x)) - \partial_x a f(x,a(x)) +  \int_{a(x)}^{b(x)} \partial_x f(x, y) \; dy . \nonumber
 \end{align}
 }, where $p_b = p(x,-H(x))$ is the bottom pressure, and integration by parts\footnote{
 Integration by parts gives:
 \begin{align}
\int_{-H}^0 z \frac{\partial p}{\partial z} \; dz & = \left[ z p \right]_{-H}^0 -   \int_{-H}^0 p \; dz , \nonumber \\ 
& = p_b H -  \int_{-H}^0 p \; dz . \nonumber
 \end{align}
 }.
 The viscous term in the $x$ direction is:
 \begin{align}
  \int_{-H}^{0} \frac{\partial}{\partial z} \left( \nu \frac{\partial u}{\partial z} \right) \; dz & = \left[  \nu \frac{\partial u}{\partial z}  \right]_{-H}^{0}  = \frac{\tau^x}{ \epsilon^2}  - \left. \nu \frac{\partial u}{\partial z} \right|_{z = -H} ,
 \end{align}
which uses the boundary condition (\ref{eqn:sfc_bcs}).
  
 Therefore (\ref{eqn:U_mom0}) and (\ref{eqn:V_mom0}) lead to:
 \begin{align}
- f V & =  - H \frac{\partial p_b}{\partial x} - \frac{\partial \chi}{\partial x}  - \tau^x_b + \tau^x , \label{eqn:U_mom} \\
f  U & = - H \frac{\partial p_b}{\partial y} - \frac{\partial \chi }{\partial y}  - \tau^y_b + \tau^y \label{eqn:V_mom} ,
\end{align}
where
\begin{align}
\chi & = - \int_{-H}^0 z b \; dz
\end{align}
is the baroclinic potential energy, and $\epsilon^2  \nu \left. \partial_z (u,v) \right|_{z = -H}  \equiv (\tau^x_b, \tau^y_b) = \tau_b$ is the bottom stress on the fluid.

Next, divide by water depth $H$:
 \begin{align}
- \frac{f V}{H} & =  - \frac{\partial p_b}{\partial x} - \frac{1}{H} \frac{\partial \chi}{\partial x} + \frac{\tau^x - \tau^x_b}{H}   \\
\frac{f  U}{H} & = - \frac{\partial p_b}{\partial y} - \frac{1}{H} \frac{\partial \chi }{\partial y} + \frac{\tau^y - \tau^y_b}{H} ,
\end{align}
and take the curl\footnote{
Recall that 
\begin{align}
{\hat{\mathbf z}} \cdot \left( \nabla \times {\mathbf U} \right) & = \partial_x V - \partial_y U .
\end{align}
}
 \begin{align}
\frac{\partial}{\partial y} \left\{- \frac{f V}{H} \right\}  - \frac{\partial}{\partial x} \left\{\frac{f  U}{H} \right\} & = 
\frac{\partial}{\partial y} \left\{ - \frac{\partial p_b}{\partial x} - \frac{1}{H} \frac{\partial \chi}{\partial x}  + \frac{\tau^x - \tau^x_b}{H} \right\} \nonumber. \\
& -  \frac{\partial}{\partial x} \left\{ - \frac{\partial p_b}{\partial y} - \frac{1}{H} \frac{\partial \chi }{\partial y}  + \frac{\tau^y - \tau^y_b}{H} \right\}, \\
\implies
\frac{\partial}{\partial y} \left\{- \frac{f }{H} \frac{\partial \Psi}{\partial x} \right\}  + \frac{\partial}{\partial x} \left\{\frac{f }{H} \frac{\partial \Psi}{\partial y} \right\} & = 
- \frac{\partial}{\partial y} \left\{ \frac{1}{H} \frac{\partial \chi}{\partial x} \right\} + \frac{\partial}{\partial x} \left\{ \frac{1}{H} \frac{\partial \chi }{\partial y}  \right\}  - {\hat{\mathbf z}} \cdot \left( \nabla \times \frac{{\mathbf \tau - \tau_b}}{H} \right), \nonumber \\
\implies
- \frac{\partial}{\partial y} \left(\frac{f }{H} \right) \, \frac{\partial \Psi}{\partial x}  + \frac{\partial}{\partial x} \, \left( \frac{f }{H}  \right) \frac{\partial \Psi}{\partial y} & = 
- \frac{\partial}{\partial y} \left( \frac{1}{H} \right) \, \frac{\partial \chi}{\partial x} + \frac{\partial}{\partial x} \, \left( \frac{1}{H} \right) \frac{\partial \chi }{\partial y}   - {\hat{\mathbf z}} \cdot \left( \nabla \times \frac{{\mathbf \tau - \tau_b}}{H} \right), \label{eqn:GCVB0} \\
\implies
J\left( \frac{f}{H} , \Psi \right)  & =  J \left( \frac{1}{H}, \chi \right)
- {\hat{\mathbf z}} \cdot \left( \nabla \times \frac{{\mathbf \tau - \tau_b}}{H} \right) ,
\label{eqn:GCVB}
\end{align}
for streamfunction of the depth-integrated horizontal flow $\Psi$ where
\begin{align}
V & = \phantom{-} \frac{\partial \Psi}{\partial x}, \nonumber \\
U & = - \frac{\partial \Psi}{\partial y} ,
\end{align}
and Jacobian $J(A,B) = \partial_x A \, \partial_y B - \partial_y A \, \partial_x B$.
Notice how this recipe (vertically integrate, divide by $H$, then take the curl) eliminates the bottom pressure.
The final result (\ref{eqn:GCVB}) is the same as Eq.~(20) of \citet{peterson&callies26}.
It is called (a simplified form of) the \textbf{geostrophic contour vorticity balance} (GCVB) by \citet{waldman&giordani23}, although \citet{peterson&callies26} call it the ``barotropic vorticity equation''.
It is not the same equation as used by \citet{nost&isachsen03} (which \citet{waldman&giordani23} call the ``transport divergence vorticity balance'').

Notice how the GCVB (\ref{eqn:GCVB}) is an equation for the streamfunction of the depth-integrated flow $\Psi$ given the buoyancy field (and hence the JEBAR term) and the surface and bottom stresses.
The GCVB shows how depth-integrated flow $\Psi$ across $f/H$ contours is driven by (i) the joint effect of baroclinicity and relief (JEBAR), (ii) curl of wind stress divided by $H$, (iii) curl of the bottom stress divided by $H$.

\section{Solution to Frictional Geostrophic Equations in Steady State}
Start from (\ref{eqn:u_mom})--(\ref{eqn:v_mom}) and write the pressure as
\begin{align}
p(x,y,z) & = p_s(x,y) + \int_{z}^{0} b(x,y,z') \; dz' , \label{eqn:hydrostatic_p} \\
\implies \partial_x p(x,y,z) & = \partial_x p_s (x, y) + \int_{z}^{0} \partial_x b(x,y,z') \; dz'  \label{eqn:hydrostatic_p_pgf} \\
\implies p_b(x, y) & \equiv p(x, y, z=-H(x,y)) = p_s(x,y) + \int_{-H(x,y)}^{0} b(x,y,z') \; dz' ,
\end{align}
where $p_s(x,y)$ is the (presently unknown) surface pressure (and assuming that $p_s \ll p(-H)$).
Solve these frictional geostrophic equations using a Green's function, defined so that $\mathfrak{u} = u + i v$, thus
\begin{align}
\epsilon^2 \frac{\partial}{\partial z} \left( \nu  \frac{\partial \mathfrak{u}}{\partial z} \right)  - i f  \mathfrak{u} & = \mathfrak{f} ,  \\
\text{Surface~}z = 0:
\epsilon^2 \nu \partial_z \mathfrak{u} & = \tau^x + i \tau^y , \label{eqn:sfc_bcs}
 \\
\text{Bottom~}z = -H:
\mathfrak{u} & = 0 ,
\end{align}
where $\mathfrak{f} = \frac{\partial p}{\partial x} +  i \frac{\partial p}{\partial y}$ is the pressure gradient from (\ref{eqn:hydrostatic_p_pgf}).

The corresponding Green's function satisfies:
\begin{align}
\epsilon^2 \frac{\partial}{\partial z} \left( \nu  \frac{\partial G_\mathfrak{u}}{\partial z} \right)  - i f  G_\mathfrak{u} & = \delta(z - \xi) ,  \\
\text{Surface~}z = 0:
\epsilon^2 \nu \partial_z G_\mathfrak{u} & = 0 ,
 \\
\text{Bottom~}z = -H:
G_\mathfrak{u} & = 0 .
\end{align}
We find that the solution for constant viscosity $\nu = \nu_0$ is:  
\begin{align}
 G_{ \mathfrak{u}}(z;\xi) & = 
\frac{ i^{3/2} \mathrm{e}^{- \sqrt{i} \varphi (z + \xi)} }{2  \nu_0 \epsilon^2 \varphi  \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\begin{cases} 
\displaystyle - \mathrm{e}^{2 \sqrt{i} \varphi \xi} + \mathrm{e}^{2 \sqrt{i} \varphi \left( z + H(x)\right)} + \mathrm{e}^{2 \sqrt{i} \varphi \left( z + \xi + H(x)\right)} - 1
 & \text{for}\: z \leq \xi , \\
\displaystyle - \mathrm{e}^{2 \sqrt{i} \varphi z} + \mathrm{e}^{2 \sqrt{i} \varphi \left( \xi + H(x)\right)} + \mathrm{e}^{2 \sqrt{i} \varphi \left( z + \xi + H(x)\right)} - 1
 & \text{for}\: z \geq \xi ,
\end{cases} , \label{eqn:FG_Gsfn} \\
\implies  G_{ \mathfrak{u}}(z;0) & = 
\frac{ i^{3/2} \mathrm{e}^{- \sqrt{i} \varphi z } }{ \nu_0 \epsilon^2 \varphi  \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right))}
\left[ \mathrm{e}^{2 \sqrt{i} \varphi ( z  + H{\left(x \right))}} - 1 \right] , \nonumber \\
\implies \partial_z \left. G_{ \mathfrak{u}}(z;\xi) \right|_{z = -H(x)} & = 
\frac{- \mathrm{e}^{-\sqrt{i} \varphi (\xi - H(x))  } }{ \nu_0 \epsilon^2 \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\left( 1 + \mathrm{e}^{2 \sqrt{i} \varphi \xi}\right) , \label{eqn:FG_Gsfn_bottom_deriv} \\
\implies \partial_z \left. G_{ \mathfrak{u}}(z;0) \right|_{z = -H(x)} & = 
\frac{- 2 \mathrm{e}^{\sqrt{i} \varphi H(x)  } }{ \nu_0 \epsilon^2 \left(1 + \mathrm{e}^{2 \sqrt{-i} \varphi H \left(x \right)} \right)}
 , \label{eqn:FG_Gsfn_bottom_deriv_top}
\end{align}
where $\varphi = \sqrt{\left( f / \nu_0 \right) } / \epsilon$ (see \texttt{Example\_barotropic\_GeostrophicFlow\_v0.2.ipynb}).
And then 
\begin{align}
\mathfrak{u} (z;x) & = \int_{-H(x)}^{0} G_{ \mathfrak{u}} (z; \xi) \left[ \partial_x p( \xi; x) + i \partial_y p( \xi; x) \right] \; d \xi 
 -   G_{\mathfrak{u}} (z; 0)  \left( \tau^x + i \tau^y \right)  ,
\end{align}
The known baroclinic pressure (due to $b$) adds to the unknown barotropic pressure (due to $p_s(x,y)$) in the integrand of this expression. 
Otherwise, everything else is known (remember that $x$ in these expressions stands for $(x,y)$ in general).

\section{Frictional Geostrophic Equations Closure with the GCVB Equation}

The problem is closed using the GCVB equation, as follows (\citealt{salmon98}, p.~170).
This determines the unknown surface pressure field given the surface stress (and vice versa).
In particular, work from (\ref{eqn:U_mom})--(\ref{eqn:V_mom}),
\begin{align}
\mathfrak{U} (z;x) & \equiv U + i V = \int_{-H(x)}^{0} \mathfrak{u} (z;x) \; dz , \label{eqn:depth_integrated_flow_def} \\
& = \int_{-H(x)}^0 \left\{ \int_{-H(x)}^{0} G_{ \mathfrak{u}} (z; \xi) \left[ \partial_x p( \xi; x) + i \partial_y p( \xi; x) \right] \; d \xi 
-  G_{\mathfrak{u}} (z; 0)  \left( \tau^x + i \tau^y \right)  \right\} \; dz.
\end{align}
Computing the $z$ integral of the Green's function $G_{\mathfrak{u}} (z; \xi)$\footnote{This presumes (correctly) that the denominator in (\ref{eqn:FG_Gsfn}) does not vanish.} gives 
\begin{align}
\int_{-H(x)}^0 G_{ \mathfrak{u}}(z;\xi) \; dz & = 
\frac{- i \mathrm{e}^{-\sqrt{i} \varphi \xi} 
\left(
\mathrm{e}^{\sqrt{i} \varphi \xi} - 
\mathrm{e}^{\sqrt{i} \varphi H(x)} 
\right) \left(
\mathrm{e}^{\sqrt{i} \varphi  \left( \xi + H(x) \right)} - 1
\right)
} { \nu_0 \epsilon^2 \varphi^2  \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H(x )} \right)} , \nonumber \\
\implies \int_{-H(x)}^0 G_{ \mathfrak{u}}(z;0) \; dz & = 
\frac{ - i \left( 
\mathrm{e}^{\sqrt{i} \varphi  H(x) }   - 1   
\right)^2
} { \nu_0 \epsilon^2 \varphi^2  \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H(x )} \right)} .
\end{align}
Therefore,
\begin{align}
\mathfrak{U} (x) & = 
\frac{-i }{{ \nu_0 \epsilon^2 \varphi^2  \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H(x )} \right)} } \times \nonumber \\
&\left\{
 \int_{-H(x)}^{0} 
\mathrm{e}^{-\sqrt{i} \varphi \xi} 
\left(
\mathrm{e}^{\sqrt{i} \varphi \xi} - 
\mathrm{e}^{\sqrt{i} \varphi H(x)} 
\right) \left(
\mathrm{e}^{\sqrt{i} \varphi  \left( \xi + H(x) \right)} - 1
\right)
 \left[ \partial_x p( \xi; x) + i \partial_y p( \xi; x) \right] \; d \xi  \right. \nonumber  \\
& - \left.
\left( 
\mathrm{e}^{\sqrt{i} \varphi  H(x) }   - 1   
\right)^2
\left( \tau^x + i \tau^y \right)   \right\}.
\end{align}
Also the bottom stress on the fluid is given by
\begin{align}
-\partial_z \left. \mathfrak{u} (z;x) \right|_{z = -H(x)} & =  \int_{-H(x)}^{0} 
 \frac{ \mathrm{e}^{-\sqrt{i} \varphi (\xi - H(x))  } }{ \nu_0 \epsilon^2 \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\left( 1 + \mathrm{e}^{2 \sqrt{i} \varphi \xi}\right) 
 \left[ \partial_x p( \xi; x) + i \partial_y p( \xi; x) \right] \; d \xi \nonumber \\
& - \frac{ 2 \mathrm{e}^{\sqrt{i} \varphi H(x)  } }{ \nu_0 \epsilon^2 \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\left( \tau^x + i \tau^y \right) , \nonumber \\
\implies - \nu_0 \epsilon^2 \partial_z \left. \mathfrak{u} (z;x) \right|_{z = -H(x)} & = \int_{-H(x)}^{0} 
\frac{ \mathrm{e}^{-\sqrt{i} \varphi (\xi - H(x))  } }{ \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\left( 1 + \mathrm{e}^{2 \sqrt{i} \varphi \xi}\right) 
 \left[ \partial_x p( \xi; x) + i \partial_y p( \xi; x) \right] \; d \xi \nonumber \\
&-\frac{ 2 \mathrm{e}^{\sqrt{i} \varphi H(x)  } }{ \left(1 + \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
 \left( \tau^x + i \tau^y \right)
 \label{eqn:bottom_stress_expression}
 \end{align}
using (\ref{eqn:FG_Gsfn_bottom_deriv}) and (\ref{eqn:FG_Gsfn_bottom_deriv_top}).
Notice how the derivatives of the unknown surface pressure part of $p$ end up multiplying expressions that are functions of $x$.

Eqns.~(\ref{eqn:U_mom}) and (\ref{eqn:V_mom}) also imply that:
\begin{align}
i f \mathfrak{U} & = - H \left(\frac{\partial p_b}{\partial x} + i \frac{\partial p_b}{\partial y} \right) - \left(\frac{\partial \chi }{\partial y} + i \frac{\partial \chi}{\partial x}  \right)  + \tau^x + i \tau^y  - \tau^x_b - i \tau^y_b .
\label{eqn:complex_vertical_int_momentum}
\end{align}
Therefore, for an unstratified fluid ($\chi = 0, \nabla p_b = \nabla p_s$), specifying the surface pressure gradient $\nabla p_s$ closes the problem. 
We compute the surface stress from (\ref{eqn:complex_vertical_int_momentum}) by substituting (\ref{eqn:bottom_stress_expression}).
Then $u(z), v(z), U, V$ and the force balance can all be computed. 
See Figure \ref{fig:example_soln} for an example calculation.
Alternatively, the surface stress can be specified, and the pressure gradient can be computed.

\section{Solution to Frictional Thermal Wind Equations in Steady State}
\label{sect:FTW_eqns}
Similarly, the frictional thermal wind equations (\ref{eqn:fric_thermal_wind}) can be solved exactly, given a $b$ field.
In that case, we need to solve
\begin{align}
- f \frac{\partial v}{\partial z} & = - \frac{\partial b}{\partial x} + \epsilon^2 \frac{\partial^2}{\partial z^2} \left( \nu \frac{\partial u}{\partial z} \right), \\
f \frac{\partial u}{\partial z} & = - \frac{\partial b}{\partial y} + \epsilon^2  \frac{\partial^2}{\partial z^2} \left( \nu  \frac{\partial v}{\partial z} \right) ,
\end{align}
subject to boundary conditions (\ref{eqn:sfc_bcs}) and (\ref{eqn:bot_bcs}).
\citet{peterson&callies26} set $\nu$ to be constant, but they also talk about the turbulent Prandtl number being one \citep{caulfield21}, so here we retain $\nu(z)$ for generality.

Write $\mathfrak{u} = u + i v$, and therefore
\begin{align}
f \frac{\partial \mathfrak{u}}{\partial z} & = - \frac{\partial b}{\partial y} + i \frac{\partial b}{\partial x} - i \epsilon^2 \frac{\partial ^2}{\partial z^2} \left( \nu \frac{\partial \mathfrak{u} }{\partial z} \right) ,  \label{eqn:FTW_fraku} \\
\text{Surface~}z = 0:
\epsilon^2 \nu \partial_z \mathfrak{u} & = \tau^x + i \tau^y,   \label{eqn:FTW_fraku_sfcbc} \\
\text{Bottom~}z = -H:
\mathfrak{u} & = 0 .
\label{eqn:FTW_fraku_botbc}
\end{align}
This is a third order linear ordinary differential equation for $\mathfrak{u}$ with two boundary conditions, which can be solved exactly, apart from an unknown integration constant.
This unknown constant corresponds to the depth-integrated current.
Alternatively, (\ref{eqn:FTW_fraku}) can be viewed as a second order equation for $\partial_z \mathfrak{u}$ with one boundary condition at the surface $z=0$. 
Therefore, it can be solved with another unknown integration constant.
Knowledge of this unknown constant would specify $\partial _z \mathfrak{u}$ at the bottom $z = -H$, which is required by the GCVB equation.
In other words, solve
\begin{align}
f \partial_z \mathfrak{u} & = - \frac{\partial b}{\partial y} + i \frac{\partial b}{\partial x} - i \epsilon^2 \frac{\partial ^2}{\partial z^2} \left( \nu \partial_z \mathfrak{u}  \right) , \nonumber \\
\text{Surface~}z = 0:
\epsilon^2 \nu \partial_z \mathfrak{u} & = \tau^x + i \tau^y,  
\end{align}
for $\partial_z \mathfrak{u}$, using a Green's function.
This problem can be solved exactly (apart from the unknown integration constant), at least for constant viscosity, and I suspect also for viscosity given like (\ref{eqn:kappa_def}). 

In detail, consider solving (\ref{eqn:FTW_fraku}) with (\ref{eqn:FTW_fraku_sfcbc}) and the alternate boundary condition
\begin{align}
\text{Bottom~}z = -H:
\epsilon^2 \nu \partial_z \mathfrak{u} & = \tau_b^x + i \tau_b^y,   
\label{eqn:FTW_fraku_botbc2}
\end{align}
where $\tau_b^x + i \tau_b^y$ is the (unknown) bottom stress.
We proceed by finding the Green's function that satisfies
\begin{align}
 i \epsilon^2  \frac{\partial^2}{\partial z^2} \left( \nu G_{\partial_z \mathfrak{u}} \right) + f  G_{\partial_z \mathfrak{u}} & = \delta(z - \xi) , \\
\text{Surface~}z = 0:
\epsilon^2 \nu G_{\partial_z \mathfrak{u}} & = 0  \\
\text{Bottom~}z = -H(x):
\epsilon^2 \nu G_{\partial_z \mathfrak{u}} & = 0 .
\end{align}
Notice how this equation looks very similar to (\ref{eqn:Gsfn})--(\ref{eqn:Gsfn_sfcbc}).
The solution for constant viscosity $\nu = \nu_0$ is:  
\begin{align}
 G_{\partial_z \mathfrak{u}} & = 
\frac{i^{3/2} \mathrm{e}^{- \sqrt{i} \varphi (z + \xi)} }{2  \nu_0 \varphi \left(1 - \mathrm{e}^{2 \sqrt{i} \varphi H \left(x \right)} \right)}
\begin{cases} 
\displaystyle \left(1 - \mathrm{e}^{2 \sqrt{i} \varphi \xi} \right) \left[1 - \mathrm{e}^{2 \sqrt{i} \varphi ( z  + H{\left(x \right))}} \right] & \text{for}\: z \leq \xi , \\
\displaystyle \left(1 - \mathrm{e}^{2 \sqrt{i} \varphi   z} \right) \left[1 - \mathrm{e}^{2 \sqrt{i} \varphi (\xi + H{\left(x \right))}} \right] & \text{for}\: z \geq \xi ,
\end{cases}
\label{eqn:FTW_Gsfn}
\end{align}
where $\varphi = \sqrt{(f / \nu_0)} / \epsilon$ (see \texttt{Example_barotropic\_GeostrophicFlow\_v0.2.ipynb}).
And then
\begin{align}
\partial_z \mathfrak{u} (z;x) & = \int_{-H(x)}^{0} G_{\partial_z \mathfrak{u}} (z; \xi) \left[ -\partial_y b( \xi; x) + i \partial_x b( \xi; x) \right] \; d \xi \nonumber \\
& +  \frac{1}{\epsilon^2 \nu_0} \left[ \partial_\xi \left. G_{\partial_z \mathfrak{u}} (z; \xi) \right|_{\xi = 0} \left( \tau^x + i \tau^y \right)  + \partial_\xi \left. G_{\partial_z \mathfrak{u}} (z; \xi) \right|_{\xi = -H(x)} \left( \tau_b^x + i \tau_b^y \right) \right] .
\label{eqn:FTW_soln}
\end{align}
We find that
\begin{align}
\partial_\xi \left. G_{\partial_z \mathfrak{u}}(z,\xi) \right|_{\xi = 0} & = 
\frac{1}{\nu_0} \frac{\left( 1 - \mathrm{e}^{2 \sqrt{i} \varphi \left( z + H(x) \right)} \right) \mathrm{e}^{- \sqrt{i} \varphi z}}
{1 - \mathrm{e}^{2 \sqrt{i} \varphi H(x)}} .
\\
\partial_\xi \left. G_{\partial_z \mathfrak{u}}(z,\xi) \right|_{\xi = -H(x)} & = 
\frac{1}{\nu_0} \frac{\left( 1 - \mathrm{e}^{2 \sqrt{i} \varphi z} \right) \mathrm{e}^{- \sqrt{i} \varphi \left( z - H(x) \right)}}
{1 - \mathrm{e}^{2 \sqrt{i} \varphi H(x)}} .
\end{align}
This procedure gives a formula for $\partial_z \mathfrak{u}$ with one unknown, namely the bottom stress $(\tau^x_b, \tau^y_b)$, and one unused boundary condition on $\mathfrak{u}$ at the bottom, namely (\ref{eqn:FTW_fraku_botbc}).

\section{Frictional Thermal Wind Equations}

The (non-dimensional) frictional thermal wind equations are therefore:
\begin{align}
- f \frac{\partial v}{\partial z} & = - \frac{\partial b}{\partial x} + \epsilon^2 \frac{\partial^2}{\partial z^2} \left( \nu \frac{\partial u}{\partial z} \right), \nonumber \\
f \frac{\partial u}{\partial z} & = - \frac{\partial b}{\partial y} + \epsilon^2 \frac{\partial^2}{\partial z^2} \left( \nu \frac{\partial v}{\partial z} \right)
\label{eqn:fric_thermal_wind}
\end{align}
(see Eqs.~(21)--(22) of \citealt{peterson&callies26}).
Boundary conditions are:
\begin{align}
\text{Surface~}z = 0:
\begin{cases}
\displaystyle \epsilon^2 \nu \partial_z u & = \tau^x \\
\displaystyle  \epsilon^2 \nu \partial_z v & = \tau^y 
\end{cases}
\label{eqn:sfc_bcs}
 \\
\text{Bottom~}z = -H:
\begin{cases}
\displaystyle u & = 0  \\
\displaystyle v & = 0 
\end{cases} ,
\label{eqn:bot_bcs}
\end{align}
where $\tau = (\tau^x, \tau^y)$ is the wind stress on the fluid.

This problem therefore has one unknown integration constant for each equation in (\ref{eqn:fric_thermal_wind}) because they are third order with two boundary conditions.
These integration constants are determined by considering the vertically-integrated flow.

Notice that (\ref{eqn:fric_thermal_wind}) can also be considered as an equation for $\partial_z (u, v)$ with one boundary condition at $z=0$.
This combination specifies $\partial_z (u,v)$ apart from an unknown integration constant.
There is also an unused condition on $(u, v)$ at $z=-H$.

\section{Solution to Buoyancy Equations in Steady State}

At least for steady state and negligible buoyancy advection, the buoyancy equation (\ref{eqn:b}) reads:
\begin{align}
\epsilon^2 \frac{\partial }{\partial z} \left( \kappa \frac{\partial b}{\partial z} \right) + \gamma \left(  B(z) - b \right) & = 0, \label{eqn:steady_vertical_b} \\
\text{Surface~}z = 0:
b &= 0
 \\
\text{Bottom~}z = -H:
\frac{\partial b}{\partial z} &= \beta
\end{align}
for buoyancy profile $B(z)$ (in principle, $B$ could depend on $(x,y)$ too) and relaxation coefficient $\gamma$, and vertical buoyancy gradient at the sea bed $\beta$ (which may be zero). 
The internal buoyancy source allows for steady-state solutions that have non-trivial solutions (the $t\rightarrow \infty$ solution to (\ref{eqn:b})--(\ref{eqn:bot_b_bc}) is vanishing $b$ everywhere; the \citealt{peterson&callies26} solutions are transient, relaxing to zero).
This problem (\ref{eqn:steady_vertical_b}) can be solved exactly using a Green's function, at least for constant diffusivity, and I suspect also for diffusivity given by (\ref{eqn:kappa_def}). 

In detail: Solve the Green's function problem first:
\begin{align}
\frac{\epsilon^2}{\gamma} \frac{\partial }{\partial z} \left( \kappa \frac{\partial G_b}{\partial z} \right) -  G_b & =  -  \delta(z - \xi) , \label{eqn:Gsfn}\\
\text{Surface~}z = 0:
G_b &= 0 ,
\label{eqn:Gsfn_sfcbc} \\
\text{Bottom~}z = -H:
\frac{\partial G_b}{\partial z} &= 0 .
\label{eqn:Gsfn_botbc}
\end{align}
This is a second order linear ordinary differential equation with two boundary conditions, which can be solved exactly.
The solution for constant diffusivity $\kappa = \kappa_0$ is:
\begin{align}
G_b (z; \xi) & = \frac{\mathrm{e}^{\phi \left(z + \xi\right)}}
{2  \kappa_0 \phi \left(\mathrm{e}^{2 \phi H \left(x \right)} + 1\right)}
\begin{cases} 
\displaystyle \left(1 - \mathrm{e}^{- 2 \phi \xi} \right) \left(\mathrm{e}^{2 \phi H{\left(x \right)}} + \mathrm{e}^{- 2 \phi z  }\right) & \text{for}\: z \leq \xi , \\
\displaystyle \left(1 - \mathrm{e}^{- 2 \phi   z} \right) \left(\mathrm{e}^{2 \phi H{\left(x \right)}} + \mathrm{e}^{- 2 \phi \xi}\right) & \text{for}\: z \geq \xi ,
\end{cases}
\end{align}
with $\phi = \sqrt{\gamma/\kappa_0} \; \epsilon$.
Therefore, given $G_b(z; \xi)$,
\begin{align}
b(z; x) = - \int_{-H(x)}^{0} G_b(z; \xi) B(\xi; x) \; d \xi  .
\end{align}
See \texttt{DiffusionEquationSolution.ipynb} and \texttt{FindBuoyancyField\_and\_GeostrophicFlow\_v0.?.ipynb}.

In what follows below, one needs horizontal derivatives of the buoyancy field.
They are:
\begin{align}
- \partial_x b(z; x) &= \partial_x H(x)\;  G_b(z; -H(x)) \; B(-H(x); x)    \nonumber \\
&+  \int_{-H(x)}^{0} \partial_x G_b(z; \xi) \; B(\xi; x) \; d \xi  \nonumber \\
&+ \int_{-H(x)}^{0} G_b(z; \xi) \; \partial_x B(\xi; x) \; d \xi  ,
\end{align}
(and similarly for the $y$ derivative--the $y$ dependence in $b, H$, and $B$ is suppressed here for simplicity).
For constant $\kappa$, $G_b(z; -H(x))$ is a relatively straightforward expression:
\begin{align}
G_b(z; -H(x)) & = 
-(\kappa_0 \phi)^{-1} \cosh \left( \phi (z + H(x)) \right) \, \tanh \left( \phi H(x) \right) 
\end{align}
and $\partial_x G_b(z; \xi)$ is:
\begin{align}
\partial_x G_b(z; \xi) & = 
\frac{\left(- \mathrm{e}^{2 z \phi} - \mathrm{e}^{2 \xi \phi} + \mathrm{e}^{2 \phi \left(z + \xi\right)} + 1\right) \mathrm{e}^{\phi \left(- z - \xi + 2 H{\left(x \right)}\right)} }{\kappa_0 \left(\mathrm{e}^{4 \phi H{\left(x \right)}} + 2 \mathrm{e}^{2 \phi H{\left(x \right)}} + 1\right)} \partial_x H{\left(x \right)} 
\end{align}
(see \texttt{FindBuoyancyField\_and\_GeostrophicFlow\_v0.4.ipynb}.).

\section{Solution Strategy for Planetary Geostrophic Equations}
Figure \ref{fig:schematic} shows a solution strategy for this set of equations. 
If we can solve a suitably-approximated buoyancy equation (or specify $b$), we can compute the baroclinic potential energy $\chi$.
We can also use this buoyancy field to force the frictional thermal wind equations.
If we can solve the frictional thermal wind equations for $\partial_z (u, v)$, up to the uncertainty left by the undetermined integration constant, we can determine the bottom stress.
Given the bottom stress and the baroclinic potential energy, we can solve the GCVB equation for the depth-integrated flow $(U,V)$ (still in terms of the undetermined integration constant).
This procedure should give the final constraint required to uniquely specify the undetermined integration constant and close the problem.

This recipe supposes that the buoyancy equation (\ref{eqn:b}) and the frictional thermal wind equations (\ref{eqn:fric_thermal_wind}) can be solved analytically. 
Under certain assumptions this is true, namely, the assumption of steady state, and the assumption of neglecting advection of buoyancy.
Then, these are both linear ordinary differential equations, so a Green's function can be used to solve them.
This procedure is explained next.

Therefore, in an abstract form, we have:
\begin{align}
U (x) & = f_1(x) \partial_x p_s + f_2(x) \partial_y p_s + f_3(x) , \nonumber \\
V (x) & = f_4(x) \partial_x p_s + f_5(x) \partial_y p_s + f_6(x) , \nonumber \\
\nu_0 \partial_z u & = b_1(x) \partial_x p_s + b_2(x) \partial_y p_s + b_3(x) , \nonumber \\
\nu_0 \partial_z v & = b_4(x) \partial_x p_s + b_5(x) \partial_y p_s + b_6(x) .
\end{align}
Thus, from (\ref{eqn:U_mom})--(\ref{eqn:V_mom})
\begin{align}
- f V & =  - H \frac{\partial p_b}{\partial x} -\frac{\partial \chi}{\partial x} - \tau^x_b + \tau^x ,  \\
f  U & = - H \frac{\partial p_b}{\partial y} - \frac{\partial \chi }{\partial y} - \tau^y_b + \tau^y ,
\end{align}
imply that
\begin{align}
g_1(x) \partial_x p_s + g_2(x) \partial_y p_s  & =  c_1(x), \\
g_3(x) \partial_x p_s + g_4(x) \partial_y p_s  & =  c_2(x) .
\end{align}
Equivalently, (\ref{eqn:GCVB0}), 
\begin{align}
\partial_ y \left(\frac{f }{H} \right)  V + \partial_ x \left( \frac{f }{H}  \right) U & = 
\partial_ y \left\{  \frac{1}{H} \frac{\partial \chi}{\partial x} + \tau^x_b - \frac{\tau^x}{H} \right\} 
-  \partial_ x \left\{ \frac{1}{H} \frac{\partial \chi }{\partial y} + \tau^y_b - \frac{\tau^y}{H} \right\}
\end{align}
implies that
\begin{align}
\left[ h_1(x) \partial_{xx}  + h_2(x) \partial_{yy}  + h_3(x) \partial_{xy} + h_4 (x) \partial_x + h_5(x) \partial_y \right] p_s & =  c_1(x) .
\end{align}
